<div class="editor-container">
    <textarea id="htmlInput" placeholder="Type your HTML here..." oninput="updateOutput()"></textarea>
    <div class="divider" onmousedown="startDragging(event)"></div>
    <div class="output-container">
        <iframe id="outputFrame"></iframe>
        <div class="placeholder" id="placeholderText">Output will be shown here...</div>
    </div>
</div>

<script>
    const htmlInput = document.getElementById("htmlInput");
    const outputFrame = document.getElementById("outputFrame");
    const placeholderText = document.getElementById("placeholderText");
    let isDragging = false;

    function updateOutput() {
        const content = htmlInput.value;
        if (content.trim() === "") {
            placeholderText.style.display = "block";
            outputFrame.srcdoc = "";
        } else {
            placeholderText.style.display = "none";
            outputFrame.srcdoc = content;
        }
    }

    function toggleMode() {
        document.body.classList.toggle("dark-mode");
    }

    function openInNewTab() {
        const newWindow = window.open();
        newWindow.document.write(htmlInput.value);
        newWindow.document.close();
    }

    function saveCode() {
        const blob = new Blob([htmlInput.value], { type: "text/html" });
        const link = document.createElement("a");
        link.href = URL.createObjectURL(blob);
        link.download = "code.html";
        link.click();
    }

    function startDragging(e) {
        isDragging = true;
        document.addEventListener("mousemove", drag);
        document.addEventListener("mouseup", stopDragging);
    }

    function drag(e) {
        if (!isDragging) return;
        const container = document.querySelector(".editor-container");
        const offsetX = e.clientX - container.offsetLeft;
        const minWidth = 100;
        const maxWidth = container.clientWidth - 100;
        if (offsetX > minWidth && offsetX < maxWidth) {
            htmlInput.style.width = offsetX + "px";
        }
    }

    function stopDragging() {
        isDragging = false;
        document.removeEventListener("mousemove", drag);
        document.removeEventListener("mouseup", stopDragging);
    }
</script>
